## Global Constraints

- **音频线程里不跑 YIN**：`pitchWorklet.ts` 只切帧转发；YIN 与分段在 Worker（P-4 spec §4.1）。
- `AudioContext` 以 `sampleRate: 48000` 创建；浏览器拒绝该选项时退回默认构造。
- `getUserMedia` 约束：`{ audio: { echoCancellation: false, noiseSuppression: false, autoGainControl: false } }`。
- 噪声基线：首次打开麦克风时校准 3 秒（`CALIBRATE_SEC = 3`），取帧 rms 的中位数。
- 只在作答阶段收音；目标音只在反馈阶段或「听目标」时播放，播放时不收音。
- 唱题（检测模式）：提示「每个音唱稳约 1 秒」；进入 answering 自动收音；唱满目标数或超时 `3 + 1.5 × 目标数` 秒自动提交；「唱完了」提前提交；本题收音出错改为自评控件。
- 判定方式缺省 `'detect'`；权限被拒 / 不支持时把 `singMode` 设为 `'self'` 并记录 `micFallbackReason`，提示一次「没有麦克风权限，唱题改为自评」或「此浏览器不支持收音，唱题改为自评」。

`src/audio/frameSlicer.ts`：

```ts
import { FRAME_SIZE, HOP_SIZE } from '../pitch/yin'

export interface SlicedFrame {
  frame: Float32Array
  /** 帧首样本的全局序号 */
  startSample: number
}

/**
 * 把音频线程送来的小块（通常 128 样本）切成 YIN 帧（2048 / hop 512）。纯函数，零 IO。
 * 跑在 AudioWorklet 里：只做拷贝与计数，不做任何分析（P-4 spec §4.1）。
 */
export function createFrameSlicer(frameSize: number = FRAME_SIZE, hop: number = HOP_SIZE) {
  const ring = new Float32Array(frameSize)
  let total = 0
  let nextEmitAt = frameSize
  return {
    push(block: Float32Array): SlicedFrame[] {
      const out: SlicedFrame[] = []
      for (let i = 0; i < block.length; i++) {
        ring[total % frameSize] = block[i]
        total++
        if (total === nextEmitAt) {
          const frame = new Float32Array(frameSize)
          const start = total - frameSize
          for (let k = 0; k < frameSize; k++) frame[k] = ring[(start + k) % frameSize]
          out.push({ frame, startSample: start })
          nextEmitAt += hop
        }
      }
      return out
    },
  }
}
```

`src/audio/pitchPipeline.ts`：

```ts
import { createSegmenter, type Segment, type Segmenter } from '../pitch/segments'
import { HOP_SIZE, yinFrame } from '../pitch/yin'

/** 噪声基线校准时长（秒），P-4 spec §4.1 */
export const CALIBRATE_SEC = 3

export type PipelineIn =
  | { type: 'calibrate-start' }
  | { type: 'calibrate-end' }
  | { type: 'capture-start'; noiseRms: number }
  | { type: 'capture-stop' }
  | { type: 'frame'; samples: Float32Array; atSec: number; sampleRate: number }

export type PipelineOut =
  | { type: 'calibrated'; noiseRms: number }
  | { type: 'segment'; segment: Segment }
  | { type: 'stopped'; segments: Segment[] }

function median(xs: number[]): number {
  if (xs.length === 0) return 0
  const s = [...xs].sort((a, b) => a - b)
  const m = s.length >> 1
  return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2
}

/** Worker 里的处理核心：纯函数式的消息处理，零 IO */
export function createPitchPipeline() {
  let mode: 'idle' | 'calibrating' | 'capturing' = 'idle'
  let rmsSamples: number[] = []
  let noiseRms = 0
  let seg: Segmenter | null = null
  let segHop = 0

  return {
    handle(msg: PipelineIn): PipelineOut[] {
      switch (msg.type) {
        case 'calibrate-start':
          mode = 'calibrating'
          rmsSamples = []
          return []
        case 'calibrate-end':
          mode = 'idle'
          return [{ type: 'calibrated', noiseRms: median(rmsSamples) }]
        case 'capture-start':
          mode = 'capturing'
          noiseRms = msg.noiseRms
          seg = null
          return []
        case 'capture-stop': {
          mode = 'idle'
          if (!seg) return [{ type: 'stopped', segments: [] }]
          const tail = seg.flush()
          const out: PipelineOut[] = tail ? [{ type: 'segment', segment: tail }] : []
          out.push({ type: 'stopped', segments: seg.segments() })
          seg = null
          return out
        }
        case 'frame': {
          const f = yinFrame(msg.samples, msg.sampleRate)
          if (mode === 'calibrating') {
            rmsSamples.push(f.rms)
            return []
          }
          if (mode !== 'capturing') return []
          const hopSec = HOP_SIZE / msg.sampleRate
          if (!seg || segHop !== hopSec) {
            seg = createSegmenter({ noiseRms, hopSec })
            segHop = hopSec
          }
          const closed = seg.push(f, msg.atSec)
          return closed ? [{ type: 'segment', segment: closed }] : []
        }
      }
    },
  }
}
```

（空闲时也会跑一次 `yinFrame`——为了代码简单。若评审认为浪费，把 `yinFrame` 调用移到 `mode` 判断之后，校准只需 rms：可直接算 rms 不跑 YIN。实现者可做这个优化，测试不变。）

`src/audio/pitchWorklet.ts`：

```ts
import { createFrameSlicer } from './frameSlicer'

/**
 * AudioWorklet 入口（浏览器专用，不单测）：只切帧转发，不做分析（P-4 spec §4.1）。
 * 由 Vite 以 `?worker&url` 打包后交给 audioWorklet.addModule。
 */
declare const sampleRate: number
declare function registerProcessor(name: string, ctor: unknown): void
declare class AudioWorkletProcessor {
  readonly port: MessagePort
}

class PitchFrameProcessor extends AudioWorkletProcessor {
  private slicer = createFrameSlicer()

  process(inputs: Float32Array[][]): boolean {
    const ch = inputs[0]?.[0]
    if (ch) {
      for (const f of this.slicer.push(ch)) {
        this.port.postMessage({ type: 'frame', samples: f.frame, atSec: f.startSample / sampleRate, sampleRate }, [f.frame.buffer])
      }
    }
    return true
  }
}

registerProcessor('pitch-frames', PitchFrameProcessor)
```

import type { Segment } from '../pitch/segments'
import { CALIBRATE_SEC, type PipelineIn, type PipelineOut } from './pitchPipeline'

export type MicFailure = 'denied' | 'unsupported' | 'error'

export class MicError extends Error {
  constructor(readonly reason: MicFailure, message: string) {
    super(message)
    this.name = 'MicError'
  }
}

export interface PortLike {
  onmessage: ((e: { data: unknown }) => void) | null
  postMessage(m: unknown, transfer?: Transferable[]): void
}
export interface WorkerLike extends PortLike {
  terminate(): void
}
export interface MicNode {
  port: PortLike
  disconnect(): void
}

export interface MicDeps {
  getUserMedia?: (c: MediaStreamConstraints) => Promise<MediaStream>
  /** audioWorklet.addModule(pitchWorklet) */
  loadWorklet(): Promise<void>
  /** MediaStreamSource → AudioWorkletNode('pitch-frames')（→ 静音增益 → destination） */
  connectSource(stream: MediaStream): MicNode
  createWorker(): WorkerLike
  wait(ms: number): Promise<void>
}

/** P-4 spec §4.1：三项浏览器处理全关，否则音高与能量会被改写 */
export const MIC_CONSTRAINTS: MediaStreamConstraints = {
  audio: { echoCancellation: false, noiseSuppression: false, autoGainControl: false },
}

export interface MicSession {
  readonly noiseRms: number
  calibrate(seconds?: number): Promise<number>
  start(onSegment: (s: Segment) => void): void
  stop(): Promise<Segment[]>
  close(): void
}

function classify(e: unknown): MicFailure {
  const name = (e as { name?: string } | null)?.name
  if (name === 'NotAllowedError' || name === 'SecurityError') return 'denied'
  if (name === 'NotFoundError' || name === 'OverconstrainedError') return 'unsupported'
  return 'error'
}

export async function openMic(deps: MicDeps): Promise<MicSession> {
  if (!deps.getUserMedia) throw new MicError('unsupported', '此浏览器不支持收音')
  let stream: MediaStream
  try {
    stream = await deps.getUserMedia(MIC_CONSTRAINTS)
  } catch (e) {
    const reason = classify(e)
    throw new MicError(reason, reason === 'denied' ? '没有麦克风权限' : `无法打开麦克风：${String((e as Error)?.message ?? e)}`)
  }
  const stopTracks = () => {
    for (const t of stream.getTracks()) t.stop()
  }
  try {
    await deps.loadWorklet()
  } catch (e) {
    stopTracks()
    throw new MicError('unsupported', `此浏览器不支持 AudioWorklet：${String((e as Error)?.message ?? e)}`)
  }

  const node = deps.connectSource(stream)
  const worker = deps.createWorker()
  const send = (m: PipelineIn, transfer?: Transferable[]) => worker.postMessage(m, transfer)

  // worklet → Worker：帧原样转发
  node.port.onmessage = (e) => {
    const m = e.data as PipelineIn
    if (m.type === 'frame') send(m, [m.samples.buffer])
  }

  let noiseRms = 0
  let onSegment: ((s: Segment) => void) | null = null
  let capturing = false
  let pending: { calibrated?: (n: number) => void; stopped?: (s: Segment[]) => void } = {}

  worker.onmessage = (e) => {
    const m = e.data as PipelineOut
    if (m.type === 'calibrated') pending.calibrated?.(m.noiseRms)
    else if (m.type === 'segment') onSegment?.(m.segment)
    else if (m.type === 'stopped') pending.stopped?.(m.segments)
  }

  return {
    get noiseRms() {
      return noiseRms
    },
    async calibrate(seconds = CALIBRATE_SEC) {
      const done = new Promise<number>((resolve) => {
        pending = { ...pending, calibrated: resolve }
      })
      send({ type: 'calibrate-start' })
      await deps.wait(seconds * 1000)
      send({ type: 'calibrate-end' })
      noiseRms = await done
      return noiseRms
    },
    start(cb) {
      onSegment = cb
      capturing = true
      send({ type: 'capture-start', noiseRms })
    },
    async stop() {
      if (!capturing) return []
      capturing = false
      const done = new Promise<Segment[]>((resolve) => {
        pending = { ...pending, stopped: resolve }
      })
      send({ type: 'capture-stop' })
      const segs = await done
      onSegment = null
      return segs
    },
    close() {
      stopTracks()
      node.disconnect()
      worker.terminate()
    },
  }
}

import { useEffect, useRef, useState } from 'react'
import type { Response } from '../../exercises/generate'
import { SING_HINT, captureTimeoutSec, type SingMic } from '../singMic'
import { PitchControl } from './PitchControl'

export function SingDetectControl(props: {
  targetsText: string | null
  targetCount: number
  enabled: boolean
  mic: SingMic
  schedule: (seconds: number, fn: () => void) => () => void
  onPlayTarget: () => void
  onResponse: (r: Response) => void
}) {
  const { enabled, mic, schedule, targetCount, onResponse } = props
  const [heard, setHeard] = useState(0)
  const [failed, setFailed] = useState(false)
  const [done, setDone] = useState(false)
  const finished = useRef(false)
  const listening = useRef(false)

  const finish = useRef<() => void>(() => {})
  finish.current = () => {
    if (finished.current) return
    finished.current = true
    setDone(true)
    listening.current = false
    mic.stop().then(
      (segs) => onResponse({ kind: 'pitch', sungHz: segs.map((s) => s.medianHz) }),
      () => setFailed(true),
    )
  }

  useEffect(() => {
    if (!enabled || finished.current) return
    let count = 0
    let cancelled = false
    listening.current = true
    mic
      .start(() => {
        if (cancelled) return
        count++
        setHeard(count)
        if (count >= targetCount) finish.current()
      })
      .catch(() => {
        listening.current = false
        if (!cancelled) setFailed(true)
      })
    const off = schedule(captureTimeoutSec(targetCount), () => finish.current())
    return () => {
      cancelled = true
      off()
      if (listening.current && !finished.current) {
        listening.current = false
        void mic.stop().catch(() => {})
      }
    }
  }, [enabled, mic, schedule, targetCount])

  if (failed) {
    return (
      <div>
        <p className="error small">收音出错，本题改为自评</p>
        <PitchControl
          targetsText={props.targetsText}
          enabled={enabled}
          onPlayTarget={props.onPlayTarget}
          onAssess={(ok) => onResponse({ kind: 'pitch', selfAssessedCorrect: ok })}
        />
      </div>
    )
  }

  return (
    <div className="pitch detect">
      <p className="hint">{SING_HINT}</p>
      {props.targetsText && <p className="targets">{props.targetsText}</p>}
      <p>{enabled ? `正在听… 已听到 ${heard} / ${targetCount} 个音` : '听完语境后开始唱'}</p>
      <button type="button" className="primary" disabled={!enabled || done} onClick={() => finish.current()}>唱完了</button>
    </div>
  )
}

import type { MicFailure, MicSession } from '../audio/micCapture'
import { MicError } from '../audio/micCapture'
import type { SingMic } from './singMic'

export interface LazyMic extends SingMic {
  readonly failure: MicFailure | null
  close(): void
}

export function createLazyMic(audio: LazyAudio, open?: (ctx: AudioContext) => Promise<MicSession>): LazyMic {
  let session: MicSession | null = null
  let opening: Promise<MicSession> | null = null
  let failure: MicFailure | null = null
  const opener = open ?? ((ctx: AudioContext) => import('./micPlatform').then((m) => m.openBrowserMic(ctx)))

  async function ensureSession(): Promise<MicSession> {
    if (session) return session
    const ctx = audio.context
    if (!ctx) throw new Error('音频尚未启用')
    if (!opening) {
      opening = (async () => {
        const s = await opener(ctx)
        await s.calibrate()
        return s
      })()
    }
    try {
      session = await opening
      failure = null
      return session
    } catch (e) {
      opening = null
      failure = e instanceof MicError ? e.reason : 'error'
      throw e
    }
  }

  return {
    get failure() {
      return failure
    },
    async ensure() {
      await ensureSession()
    },
    async start(onSegment) {
      ;(await ensureSession()).start(onSegment)
    },
    async stop() {
      return session ? session.stop() : []
    },
    close() {
      session?.close()
      session = null
      opening = null
    },
  }
}

/** 浏览器专用：真实的 getUserMedia / AudioWorklet / Worker。只由 platform.ts 动态 import */
export function openBrowserMic(ctx: AudioContext): Promise<MicSession> {
  const md = typeof navigator !== 'undefined' ? navigator.mediaDevices : undefined
  return openMic({
    getUserMedia: md?.getUserMedia ? (c) => md.getUserMedia(c) : undefined,
    loadWorklet: async () => {
      if (!ctx.audioWorklet || typeof AudioWorkletNode === 'undefined') throw new Error('AudioWorklet 不可用')
      await ctx.audioWorklet.addModule(workletUrl)
    },
    connectSource: (stream) => {
      const source = ctx.createMediaStreamSource(stream)
      const node = new AudioWorkletNode(ctx, 'pitch-frames')
      const sink = ctx.createGain()
      sink.gain.value = 0 // 让节点保持被拉动，但不出声
      source.connect(node)
      node.connect(sink).connect(ctx.destination)
      return {
        port: node.port as unknown as WorkerLike,
        disconnect: () => {
          source.disconnect()
          node.disconnect()
          sink.disconnect()
        },
      }
    },
    createWorker: () => new Worker(new URL('../audio/pitchWorker.ts', import.meta.url), { type: 'module' }) as unknown as WorkerLike,
    wait: (ms) => new Promise((r) => setTimeout(r, ms)),

import { useRef, useState } from 'react'
import { VOICE_PRESETS, type VocalRange, type VoicePreset } from '../../pitch/range'
import { pickHeldNote, validateRange } from '../../pitch/rangeTest'
import { useRuntime } from '../RuntimeContext'

export const HOLD_SEC = 4

const NOTE_NAMES = ['C', 'C♯', 'D', 'D♯', 'E', 'F', 'F♯', 'G', 'G♯', 'A', 'A♯', 'B']
export function midiName(m: number): string {
  return `${NOTE_NAMES[((m % 12) + 12) % 12]}${Math.floor(m / 12) - 1}`
}

type Step =
  | { kind: 'intro' }
  | { kind: 'presets' }
  | { kind: 'calibrating' }
  | { kind: 'hold'; which: 'low' | 'high'; low: number | null }
  | { kind: 'result'; range: VocalRange }
  | { kind: 'invalid'; reason: string }

const PRESET_ORDER: VoicePreset[] = ['bass', 'tenor', 'alto', 'soprano']

export function RangeTest(props: { onDone?: () => void }) {
  const { runtime, audio, mic } = useRuntime()
  const [step, setStep] = useState<Step>({ kind: 'intro' })
  const [error, setError] = useState<string | null>(null)
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null)
  const msg = (e: unknown) => (e instanceof Error ? e.message : String(e))

  async function save(vocalRange: VocalRange | null) {
    try {
      await runtime.updateProfile({ vocalRange })
      props.onDone?.()
    } catch (e) {
      setError(msg(e))
    }
  }

  async function startHold(which: 'low' | 'high', low: number | null) {
    setStep({ kind: 'hold', which, low })
    try {
      await mic.start(() => {})
      timer.current = setTimeout(() => void finishHold(which, low), HOLD_SEC * 1000)
    } catch (e) {
      setError(msg(e))
      setStep({ kind: 'intro' })
    }
  }

  async function finishHold(which: 'low' | 'high', low: number | null) {
    if (timer.current) clearTimeout(timer.current)
    timer.current = null
    try {
      const note = pickHeldNote(await mic.stop())
      if (which === 'low') {
        await startHold('high', note)
        return
      }
      const check = validateRange(low, note)
      setStep(check.ok ? { kind: 'result', range: check.range } : { kind: 'invalid', reason: check.reason })
    } catch (e) {
      setError(msg(e))
      setStep({ kind: 'intro' })
    }
  }

  async function begin() {
    setError(null)
    try {
      await audio.unlock()
      setStep({ kind: 'calibrating' })
      await mic.ensure()
      await startHold('low', null)
    } catch (e) {
      setError(msg(e))
      setStep({ kind: 'intro' })
    }
  }

  const errorLine = error && <p className="error">{error}</p>

  switch (step.kind) {
    case 'intro':
      return (
        <main className="page onboarding">
          <h1>测一下你的音域</h1>
          <p>唱题会落在你唱得舒服的地方。需要麦克风，约 30 秒。</p>
          {errorLine}
          <button type="button" className="primary big" onClick={() => void begin()}>开始测定</button>
          <button type="button" onClick={() => setStep({ kind: 'presets' })}>选声部</button>
          <button type="button" onClick={() => void save(null)}>跳过，以后再测</button>
        </main>
      )
    case 'presets':
      return (
        <main className="page onboarding">
          <h1>选一个最接近你的声部</h1>
          {errorLine}
          <div className="row">
            {PRESET_ORDER.map((p) => (
              <button key={p} type="button" onClick={() => void save({ lowMidi: VOICE_PRESETS[p].lowMidi, highMidi: VOICE_PRESETS[p].highMidi })}>
                {VOICE_PRESETS[p].label}
              </button>
            ))}
          </div>
          <button type="button" onClick={() => setStep({ kind: 'intro' })}>返回</button>
        </main>
      )
    case 'calibrating':
      return (
        <main className="page onboarding">
          <p>正在校准环境噪声，请保持安静…</p>
        </main>
      )
    case 'hold':
      return (
        <main className="page onboarding">
          <h1>{step.which === 'low' ? '唱一个你舒服的最低音并保持' : '再唱一个你舒服的最高音并保持'}</h1>
          <p>用「啊」唱，保持 2 秒左右。</p>
          <button type="button" className="primary big" onClick={() => void finishHold(step.which, step.low)}>唱好了</button>
        </main>
      )
    case 'result':
      return (
        <main className="page onboarding">
          <h1>你的音域：{midiName(step.range.lowMidi)} – {midiName(step.range.highMidi)}</h1>
          {errorLine}
          <button type="button" className="primary big" onClick={() => void save(step.range)}>保存</button>
          <button type="button" onClick={() => void begin()}>重新测定</button>
        </main>
      )
    case 'invalid':
      return (
        <main className="page onboarding">
          <p className="error">{step.reason}</p>
          <button type="button" className="primary big" onClick={() => void begin()}>重新测定</button>
          <button type="button" onClick={() => setStep({ kind: 'presets' })}>选声部</button>
        </main>
      )
  }
}
